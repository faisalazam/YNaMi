119     /*
120      * This method will return a collection of 'Set<Role>', where each 'Set<Role>' will have different size.
121      */
122     @SuppressWarnings("OptionalGetWithoutIsPresent")
123     protected Collection<Set<Role>> getRolesCollection(final RoleRepository roleRepository) {
124         final List<Role> allRoles = stream(values())
125                 .map(roleName -> roleRepository.findByName(roleName).get())
126                 .collect(toList());
127         assertThat(allRoles.size(), is(values().length));
128 
129         final List<Set<Role>> associatedRolesList = newArrayList();
130         for (int i = 1; i <= allRoles.size(); i++) {
131             associatedRolesList.add(newHashSet(allRoles.subList(0, i)));
132         }
133         assertThat(associatedRolesList.size(), is(values().length));
134 
135         return associatedRolesList;
136     }
