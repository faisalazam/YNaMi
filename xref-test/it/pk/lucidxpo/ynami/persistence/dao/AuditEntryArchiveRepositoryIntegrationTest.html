18  class AuditEntryArchiveRepositoryIntegrationTest extends AbstractIntegrationTest {
19      @Autowired
20      private AuditEntryArchiveRepository archiveRepository;
21  
22      @Test
23      void shouldRetrieveAuditEntryArchiveRecordsByEntityId() {
24  
25          final String matchingEntityId = randomID();
26          final String nonMatchingEntityId = randomID();
27  
28          final AuditEntry auditEntry1 = new AuditEntry(randomID(), matchingEntityId, "testField1",
29                  "fromValue1", "toValue1", "changedBy1");
30          final AuditEntryArchive auditEntryArchive1 = new AuditEntryArchive();
31          copyProperties(auditEntry1, auditEntryArchive1);
32  
33          final AuditEntry auditEntry2 = new AuditEntry(randomID(), matchingEntityId, "testField2",
34                  "fromValue2", "toValue2", "changedBy2");
35          final AuditEntryArchive auditEntryArchive2 = new AuditEntryArchive();
36          copyProperties(auditEntry2, auditEntryArchive2);
37  
38          final AuditEntry auditEntry3 = new AuditEntry(randomID(), nonMatchingEntityId, "testField3",
39                  "fromValue3", "toValue3", "changedBy3");
40          final AuditEntryArchive auditEntryArchive3 = new AuditEntryArchive();
41          copyProperties(auditEntry3, auditEntryArchive3);
42  
43          archiveRepository.save(auditEntryArchive1);
44          archiveRepository.save(auditEntryArchive2);
45          archiveRepository.save(auditEntryArchive3);
46  
47          List<AuditEntryArchive> records = archiveRepository.findByChangedEntityIdOrderByChangedAtDesc(matchingEntityId);
48          assertEquals(2, records.size());
49          assertEntityAttributes(auditEntryArchive2, records.get(0));
50          assertEntityAttributes(auditEntryArchive1, records.get(1));
51  
52          records = archiveRepository.findByChangedEntityIdOrderByChangedAtDesc(nonMatchingEntityId);
53          assertEquals(1, records.size());
54          assertEntityAttributes(auditEntryArchive3, records.get(0));
55  
56          records = archiveRepository.findByChangedEntityIdOrderByChangedAtDesc(randomID());
57          assertTrue(records.isEmpty());
58      }
59  
60      private void assertEntityAttributes(final AuditEntryArchive expectedAuditEntryArchive, final AuditEntryArchive actualAuditEntryArchive) {
61          assertAll(
62                  () -> assertEquals(expectedAuditEntryArchive.getChangedEntityName(), actualAuditEntryArchive.getChangedEntityName()),
63                  () -> assertEquals(expectedAuditEntryArchive.getChangedEntityId(), actualAuditEntryArchive.getChangedEntityId()),
64  
65                  () -> assertEquals(expectedAuditEntryArchive.getFieldChanged(), actualAuditEntryArchive.getFieldChanged()),
66                  () -> assertEquals(expectedAuditEntryArchive.getFromValue(), actualAuditEntryArchive.getFromValue()),
67                  () -> assertEquals(expectedAuditEntryArchive.getToValue(), actualAuditEntryArchive.getToValue()),
68  
69                  () -> assertEquals(expectedAuditEntryArchive.getChangedBy(), actualAuditEntryArchive.getChangedBy()),
70                  () -> assertEquals(expectedAuditEntryArchive.getChangedAt(), actualAuditEntryArchive.getChangedAt())
71          );
72      }
