17  @TestPropertySource(properties = "config.togglz.enabled=true")
18  class PersistenceConfigIntegrationTest extends AbstractIntegrationTest {
19      @Test
20      void shouldVerifyThatAuditorAwareBeanIsAnInstanceOfSpringSecurityAuditAwareImplWhenWebSecurityIsEnabled() throws Exception {
21          final String auditorAwareBeanName = (String) getField(PersistenceConfig.class, null, "AUDITOR_AWARE_BEAN_NAME");
22          final InvocationHandler invocationHandler = getInvocationHandler(applicationContext.getBean(auditorAwareBeanName));
23          final Object auditorAwareBean = invocationHandler.getClass().getDeclaredMethod("getActive").invoke(invocationHandler);
24          assertThat(auditorAwareBean, instanceOf(SpringSecurityAuditAwareImpl.class));
25      }
26  
27      @Test
28      void shouldVerifyThatAuditorAwareBeanIsAnInstanceOfAuditorAwareImplWhenWebSecurityIsDisabled() throws Exception {
29          final String auditorAwareBeanName = (String) getField(PersistenceConfig.class, null, "AUDITOR_AWARE_BEAN_NAME");
30          final InvocationHandler invocationHandler = getInvocationHandler(applicationContext.getBean(auditorAwareBeanName));
31          final Object auditorAwareBean = invocationHandler.getClass().getDeclaredMethod("getInactive").invoke(invocationHandler);
32          assertThat(auditorAwareBean, instanceOf(AuditorAwareImpl.class));
33      }
