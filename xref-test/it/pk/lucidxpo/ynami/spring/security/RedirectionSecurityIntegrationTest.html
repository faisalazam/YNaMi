46      @Test
47      @EnabledIf(value = "${togglz.features.WEB_SECURITY.enabled:false}", loadContext = true)

85      @Test
86      @EnabledIf(value = "${togglz.features.WEB_SECURITY.enabled:false}", loadContext = true)
87      void shouldVerifyRedirectedBackToSecuredResourceAfterAuthenticationWhenSecuredResourceIsAccessedUnauthenticated() throws Exception {
88          final MockHttpServletRequestBuilder securedResourceAccess = get("/samples");
89          final MvcResult unauthenticatedResult = mockMvc.perform(securedResourceAccess)
90                  .andExpect(status().is3xxRedirection())
91                  .andExpect(redirectedUrlPattern("**" + LOGIN_PAGE_URL))
92                  .andExpect(unauthenticated())
93                  .andReturn();
94  
95          final MockHttpSession session = (MockHttpSession) unauthenticatedResult.getRequest().getSession();
96          assertNotNull(session);
97  
98          final String loginProcessingUrl = (String) getField(SecurityConfig.class, null, "LOGIN_PROCESSING_URL");
99          mockMvc.perform(post(loginProcessingUrl)
100                         .param("username", "admin")
101                         .param("password", "admin")
102                         .session(session)
103                         .with(csrf()))
104                 .andExpect(status().is3xxRedirection())
105                 .andExpect(redirectedUrlPattern("**/samples?continue"))
106                 .andExpect(authenticated())
107                 .andReturn();
108 
109         mockMvc.perform(securedResourceAccess.session(session))
110                 .andExpect(status().isOk())
111                 .andExpect(authenticated());
112     }

114     private static ResultActions assertHeadersExcludingXFrameOptions(ResultActions actions) throws Exception {
115         return actions
116                 .andExpect(header().string("X-Content-Type-Options", is(NOSNIFF)))
117                 .andExpect(header().string("X-XSS-Protection", notNullValue()))
118                 .andExpect(header().string("Cache-Control", notNullValue()))
119                 .andExpect(header().string("Pragma", is(NO_CACHE.asString())));
120     }
