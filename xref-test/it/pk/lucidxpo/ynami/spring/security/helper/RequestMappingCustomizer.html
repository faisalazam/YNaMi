27  /*
28   * RequestMappingCustomizer: This class is responsible for customizing the request mappings.
29   *
30   * It'll look at the request url template and figure out if there are any path variables, and if so, then it'll fill
31   * them with random values.
32   *
33   * If it involves any entity, then it'll create the Pojo for that entity filled with the random data and save it in the
34   * database using that entity's corresponding JPA repository.'
35   *
36   * Then it'll fill the 'resultActions' object with the matchers from the 'resultMatchers' property which will be used
37   * later on to perform assertions.
38   */
39  public class RequestMappingCustomizer {
40      private final Class<?> entityClazz;
41      private final String methodType;
42      private final String urlTemplate;
43      private Pair<String, Object> modelAttribute;
44      private final ResultMatcher[] resultMatchers;
45      private final MockHttpServletRequestBuilder requestBuilder;
46  
47      public RequestMappingCustomizer(final String methodType,
48                                      final String urlTemplate,
49                                      final ResultMatcher... resultMatchers) {
50          this.entityClazz = null;
51          this.requestBuilder = null;
52          this.methodType = methodType;
53          this.urlTemplate = urlTemplate;
54          this.resultMatchers = resultMatchers;
55      }

57      public RequestMappingCustomizer(final String methodType,

60                                      final ResultMatcher... resultMatchers) {
61          this.requestBuilder = null;
62          this.methodType = methodType;
63          this.entityClazz = entityClazz;
64          this.urlTemplate = urlTemplate;
65          this.resultMatchers = resultMatchers;
66      }

68      public RequestMappingCustomizer(final String methodType,
69                                      final MockHttpServletRequestBuilder requestBuilder,
70                                      final ResultMatcher... resultMatchers) {
71          this.entityClazz = null;
72          this.urlTemplate = EMPTY;
73          this.methodType = methodType;
74          this.requestBuilder = requestBuilder;
75          this.resultMatchers = resultMatchers;
76      }

78      /**
79       * This method will fill the 'resultActions' object with the matchers from the 'resultMatchers' property.
80       *
81       * @param resultActions: the result actions object to contain the expectations or assertions, which will be executed
82       *                       later on.
83       */
84      void assertExpectations(final ResultActions resultActions) throws Exception {
85          if (isNotEmpty(resultMatchers)) {
86              for (final ResultMatcher resultMatcher : resultMatchers) {
87                  resultActions.andExpect(resultMatcher);
88              }
89          } else {
90              resultActions.andExpect(status().isOk());
91          }
92      }
93  
94      /**
95       * This method is used to return configured MockHttpServletRequestBuilder object which will be used to build the request.
96       *
97       * @return MockHttpServletRequestBuilder
98       */
99      MockHttpServletRequestBuilder getRequestBuilder(final ApplicationContext applicationContext) {
100         final MockHttpServletRequestBuilder requestBuilder;
101         if (this.requestBuilder != null) {
102             requestBuilder = this.requestBuilder;
103         } else if (entityClazz != null) {
104             final Identifiable identifiable = saveEntity(applicationContext);
105             requestBuilder = request(valueOf(methodType), urlTemplate, identifiable.getId());
106         } else {
107             requestBuilder = request(valueOf(methodType), urlTemplate, uriVariables(urlTemplate));
108         }
109 
110         if (modelAttribute != null) {
111             requestBuilder.flashAttr(modelAttribute.getKey(), modelAttribute.getValue());
112         }
113         return requestBuilder;
114     }
115

117         return methodType;

124     /*
125      * This method is called to build an array of objects containing the values for the path variables present in the
126      * @param urlTemplate. The @param urlTemplate may or may not contain path variables.
127      */
128     static Object[] uriVariables(final String urlTemplate) {
129         final Pattern pathVariablesCapturingPattern = compile("\\{(.*?)}");
130         final Matcher matcher = pathVariablesCapturingPattern.matcher(urlTemplate);
131         final int numberOfPathVariables = (int) matcher.results().count();
132         return range(0, numberOfPathVariables).mapToObj(i -> randomID()).toArray();
133     }

135     @SuppressWarnings({"unchecked", "rawtypes"})
136     /*
137      * This method is used to create a new instance of 'entityClazz' filled with random data using Podam.
138      * Then it does retrieve the corresponding JPA repository from the application context for the specified entityClazz.
139      * And finally uses the retrieved repository to save the newly created instance in db.
140      *
141      * entityClazz: Class, of which Pojo will be created and filled with random data.
142      */
143     private Identifiable saveEntity(final ApplicationContext applicationContext) {
144         // PODAM is a lightweight tool to autofill Java POJOs with data. So, the following statement will create an
145         // instance of 'entityClazz' filled with random data.
146         final Object entity = new PodamFactoryImpl().manufacturePojo(entityClazz);
147         assertNotNull(entity);
148 
149         final String entitySimpleName = requireNonNull(entityClazz).getSimpleName();
150         final String entityRepositoryName = uncapitalize(entitySimpleName) + "Repository";
151         final JpaRepository repository = (JpaRepository) applicationContext.getBean(entityRepositoryName);
152         return (Identifiable) repository.saveAndFlush(entity);
153     }
