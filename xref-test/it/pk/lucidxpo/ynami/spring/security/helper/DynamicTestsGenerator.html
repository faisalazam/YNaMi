4   import org.junit.jupiter.api.DynamicTest;
5   import org.springframework.context.ApplicationContext;
6   import org.springframework.test.web.servlet.MockMvc;
7   import pk.lucidxpo.ynami.spring.features.FeatureManagerWrappable;
8   
9   import java.util.Collection;
10  
11  import static it.pk.lucidxpo.ynami.spring.security.helper.EndPointTestScenarioExecutor.performAndAssertExpectations;
12  import static java.util.stream.Collectors.toList;
13  import static org.junit.jupiter.api.Assumptions.assumeFalse;
14  import static org.junit.jupiter.api.Assumptions.assumeTrue;
15  import static org.junit.jupiter.api.DynamicTest.dynamicTest;
16  import static pk.lucidxpo.ynami.spring.features.FeatureToggles.WEB_SECURITY;
17  
18  public class DynamicTestsGenerator {
19      private final ApplicationContext applicationContext;
20      private final FeatureManagerWrappable featureManager;
21      private final Collection<Pair<String, RequestMappingCustomizer>> endPointMappingsStream;
22  
23      public DynamicTestsGenerator(final ApplicationContext applicationContext,
24                                   final FeatureManagerWrappable featureManager,
25                                   final Collection<Pair<String, RequestMappingCustomizer>> endPointMappingsStream) {
26          this.featureManager = featureManager;
27          this.applicationContext = applicationContext;
28          this.endPointMappingsStream = endPointMappingsStream;
29      }
30  
31      /**
32       * @return Generates and returns a collection of dynamic tests by going through the stream of endpoint mappings.
33       */
34      public Collection<DynamicTest> getDynamicTestsWhenWebSecurityIsEnabled(final MockMvc mockMvc,
35                                                                             final EndPointTestScenario endPointTestScenario) {
36          return endPointMappingsStream.stream()
37                  .map(requestBuilderPair -> dynamicTest(requestBuilderPair.getKey(), () -> {
38                      assumeTrue(featureManager.isActive(WEB_SECURITY),
39                              requestBuilderPair.getKey() + " test is ignored as Web Security is disabled"
40                      );
41  
42                      performAndAssertExpectations(
43                              mockMvc,
44                              applicationContext,
45                              requestBuilderPair.getValue(),
46                              endPointTestScenario
47                      );
48                  }))
49                  .collect(toList());
50      }
51  
52      /**
53       * @return Generates and returns a collection of dynamic tests by going through the stream of endpoint mappings.
54       */
55      public Collection<DynamicTest> getDynamicTestsWhenWebSecurityIsDisabled(final MockMvc mockMvc,
56                                                                              final EndPointTestScenario endPointTestScenario) {
57          return endPointMappingsStream.stream()
58                  .map(requestBuilderPair -> dynamicTest(requestBuilderPair.getKey(), () -> {
59                      assumeFalse(featureManager.isActive(WEB_SECURITY),
60                              requestBuilderPair.getKey() + " test is ignored as Web Security is enabled"
61                      );
62  
63                      performAndAssertExpectations(
64                              mockMvc,
65                              applicationContext,
66                              requestBuilderPair.getValue(),
67                              endPointTestScenario
68                      );
69                  }))
70                  .collect(toList());
71      }
