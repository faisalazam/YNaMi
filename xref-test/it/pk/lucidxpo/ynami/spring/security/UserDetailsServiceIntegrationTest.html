47  @TestExecutionListeners(value = DatabaseExecutionListener.class, mergeMode = MERGE_WITH_DEFAULTS)
48  class UserDetailsServiceIntegrationTest extends AbstractIntegrationTest {
49      @Autowired
50      private RoleRepository roleRepository;
51  
52      @Autowired
53      private UserRepository userRepository;
54  
55      @Autowired
56      @Qualifier("userDetailsServiceImpl")
57      private UserDetailsService userDetailsService;
58  
59      @BeforeEach
60      void setup() {
61          featureManager.activate(WEB_SECURITY);
62          assertTrue(featureManager.isActive(WEB_SECURITY));
63      }
64  
65      @Test
66      void shouldThrowUsernameNotFoundExceptionOnLoadUserByUsernameWhenUserWithSpecifiedUsernameOrEmailDoesNotExist() {
67          final String usernameOrEmail = randomAlphanumeric(5, 35);
68          try {
69              userDetailsService.loadUserByUsername(usernameOrEmail);
70              fail("Should have thrown UsernameNotFoundException as there is no user with specified username or email: " + usernameOrEmail);
71          } catch (final Exception e) {
72              assertThat(e, instanceOf(UsernameNotFoundException.class));
73          }
74      }
75  
76      @Test
77      void shouldGetUserDetailsOnLoadUserByUsernameWhenUserWithSpecifiedUsernameDoesExist() {
78          final String username = randomAlphanumeric(5, 35);
79          final User user = anUser().withUsername(username).build();
80  
81          saveAndAssertUserDetails(username, user);
82      }
83  
84      @Test
85      void shouldGetUserDetailsOnLoadUserByUsernameWhenUserWithSpecifiedEmailDoesExist() {
86          final String email = randomAlphanumeric(5) + "@" + randomAlphanumeric(5) + "." + randomAlphanumeric(3);
87          final User user = anUser().withEmail(email).build();
88  
89          saveAndAssertUserDetails(email, user);
90      }
91  
92      private void saveAndAssertUserDetails(final String usernameOrEmail, final User user) {
93          final Set<Role> roles = chooseOneOf(getRolesCollection(roleRepository));
94          user.setRoles(roles);
95  
96          final User savedUser = userRepository.save(user);
97          assertAuditInfo(savedUser, ADMIN_USER);
98  
99          final UserPrincipal userDetails = (UserPrincipal) userDetailsService.loadUserByUsername(usernameOrEmail);
100         final List<? extends GrantedAuthority> expectedAuthorities = roles.stream()
101                 .map(role -> new SimpleGrantedAuthority(role.getName().name()))
102                 .collect(toList());
103         assertUserDetails(savedUser, userDetails, expectedAuthorities);
104     }
105 
106     private void assertUserDetails(final User expectedUserDetails,
107                                    final UserPrincipal actualUserDetails,
108                                    final List<? extends GrantedAuthority> expectedAuthorities) {
109         assertAll(
110                 () -> assertEquals(expectedUserDetails.getId(), actualUserDetails.getId()),
111                 () -> assertEquals(expectedUserDetails.getName(), actualUserDetails.getName()),
112                 () -> assertEquals(expectedUserDetails.getUsername(), actualUserDetails.getUsername()),
113                 () -> assertEquals(expectedUserDetails.getEmail(), actualUserDetails.getEmail()),
114                 () -> assertEquals(expectedUserDetails.getPassword(), actualUserDetails.getPassword()),
115                 () -> assertTrue(actualUserDetails.isAccountNonExpired()),
116                 () -> assertTrue(actualUserDetails.isAccountNonLocked()),
117                 () -> assertTrue(actualUserDetails.isCredentialsNonExpired()),
118                 () -> assertTrue(actualUserDetails.isEnabled()),
119                 () -> assertTrue(actualUserDetails.getAuthorities().containsAll(expectedAuthorities))
120         );
121     }
