10  public class DBTableColumnMetaDataUtil {
11  
12      @SuppressWarnings("ConstantConditions")
13      public static List<DBTableColumnMetaData> getColumnMetaData(final MultiSqlExecutor executor, final String tableName) {
14          final String querySQL = "SELECT TABLE_NAME, COLUMN_NAME, DATA_TYPE, CHARACTER_MAXIMUM_LENGTH, "
15                  + "NUMERIC_PRECISION, DATETIME_PRECISION, IS_NULLABLE, COLUMN_DEFAULT "
16                  + "FROM INFORMATION_SCHEMA.COLUMNS "
17                  + "WHERE upper(TABLE_SCHEMA) = '" + SCHEMA_NAME.toUpperCase() + "' "
18                  + "AND upper(TABLE_NAME) = '" + tableName.toUpperCase() + "' ";
19          return executor.getTemplate().query(querySQL, null, null, (rs, rowNum) -> new DBTableColumnMetaData(
20                  rs.getString("TABLE_NAME"),
21                  rs.getString("COLUMN_NAME"),
22                  rs.getString("DATA_TYPE"),
23                  rs.getBigDecimal("CHARACTER_MAXIMUM_LENGTH"),
24                  rs.getInt("NUMERIC_PRECISION"),
25                  rs.getInt("DATETIME_PRECISION"),
26                  toBoolean(rs.getString("IS_NULLABLE")),
27                  rs.getString("COLUMN_DEFAULT")
28          ));
29      }
30  
31      public static void assertColumnExists(final List<DBTableColumnMetaData> tableColumnMetaData,
32                                            final String columnName,
33                                            final String dataType) {
34          for (final DBTableColumnMetaData columnMetaData : tableColumnMetaData) {
35              if (columnName.equalsIgnoreCase(columnMetaData.getColumnName())
36                      && getConvertedDataType(dataType).equalsIgnoreCase(columnMetaData.getDataType())) {
37                  return;
38              }
39          }
40          fail("The table [" + tableColumnMetaData.get(0).getTableName() + "] should have column ["
41                  + columnName + "] with dataType [" + dataType + "]"
42          );
43      }
44  }
