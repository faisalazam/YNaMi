39      public static DataSource localDataSource() {
40          final String dataSourceUrl, userName, password;
41          if (CONNECT_TO_MYSQL) {
42              userName = "root";
43              password = "root";
44              dataSourceUrl = "jdbc:mysql://localhost:3306/" + SCHEMA_NAME +
45                      "?createDatabaseIfNotExist=true" +
46                      "&useSSL=false&allowPublicKeyRetrieval=true";
47          } else {
48              userName = "sa";
49              password = EMPTY_STRING;
50              dataSourceUrl = "jdbc:h2:file:./h2_db/" + SCHEMA_NAME +
51                      ";MODE=MySQL" +
52                      ";AUTO_SERVER=TRUE" +
53                      ";DB_CLOSE_DELAY=-1" +
54                      ";DB_CLOSE_ON_EXIT=TRUE" +
55                      ";DATABASE_TO_UPPER=FALSE" +
56                      ";INIT=CREATE SCHEMA IF NOT EXISTS " + SCHEMA_NAME + "\\" +
57                      ";SET SCHEMA " + SCHEMA_NAME;
58          }
59          return new DriverManagerDataSource(dataSourceUrl, userName, password);
60      }

119     public static void assertColumnWith(final MultiSqlExecutor executor,
120                                         final String tableName,
121                                         final String columnName,
122                                         final String dataType,
123                                         final boolean nullable,
124                                         final long size) {
125 
126         String query = "SELECT * FROM INFORMATION_SCHEMA.COLUMNS "
127                 + "WHERE upper(TABLE_SCHEMA) = '" + SCHEMA_NAME.toUpperCase() + "' "
128                 + "AND upper(TABLE_NAME) = '" + tableName.toUpperCase() + "' "
129                 + "AND upper(COLUMN_NAME) = '" + columnName.toUpperCase() + "' ";
130         System.out.println("\nCheck if column exists with specifics: \n" + query);
131 
132         final Map<String, Object> result = executor.getTemplate().queryForMap(query);
133         assertFalse(result.isEmpty());
134         assertEquals((nullable ? "YES" : "NO"), result.get("IS_NULLABLE"));
135 
136         final String convertedDataType = getConvertedDataType(dataType);
137         assertThat(result.get("DATA_TYPE").toString(), equalToIgnoringCase(convertedDataType));
138 
139         final String sizeColumnName = getSizeColumnName(dataType);
140         final Object sizeObject = result.get(sizeColumnName);
141         if (sizeObject instanceof Long) {
142             assertEquals(size, sizeObject);
143         } else {
144             if (DATA_TYPE_H2_BIT.equalsIgnoreCase(convertedDataType)) {
145                 // H2 has no size for this type
146                 assertNull(sizeObject);
147             } else {
148                 // this is just for H2 compatibility as MySql will return the size in long always
149                 // but H2 will return the size in integer for DateTime columns and in long otherwise.
150                 assertEquals(((Long) size).intValue(), sizeObject);
151             }
152         }
153     }

155     public static String getSizeColumnName(final String dataType) {
156         if (dataType.toLowerCase().contains("int") || dataType.toLowerCase().contains("bit")) {
157             return "NUMERIC_PRECISION";
158         } else if (dataType.toLowerCase().contains("timestamp")) {
159             return "DATETIME_PRECISION";
160         } else {
161             return "CHARACTER_MAXIMUM_LENGTH";
162         }

165     public static void assertConstraintExistsForTable(final MultiSqlExecutor executor,
166                                                       final String tableName,
167                                                       final String constraintType,
168                                                       final String constraintName) {
169         final String query = "SELECT * FROM INFORMATION_SCHEMA.TABLE_CONSTRAINTS "
170                 + "WHERE upper(TABLE_SCHEMA) = '" + SCHEMA_NAME.toUpperCase() + "' "
171                 + "AND upper(TABLE_NAME) = '" + tableName.toUpperCase() + "' ";
172         System.out.println("\nCheck if table exists with constraint: \n" + query);
173 
174         final List<Map<String, Object>> constraints = executor.getTemplate().queryForList(query);
175         final String convertedConstraintName = getConvertedConstraintName(constraintType, constraintName);
176         final long count = constraints
177                 .stream()
178                 .filter(map ->
179                         map.get("CONSTRAINT_TYPE").toString().equalsIgnoreCase(constraintType) &&
180                                 map.get("CONSTRAINT_NAME").toString().equalsIgnoreCase(convertedConstraintName)
181                 ).count();
182         assertEquals(1, count,
183                 "Found " + constraints.size() + " but none of them matched with specified attributes");
184     }

203     public static String getConvertedDataType(final String dataType) {
204         if (CONNECT_TO_MYSQL) {
205             return dataType;
206         }
207         if (DATA_TYPE_VARCHAR.equalsIgnoreCase(dataType) || DATA_TYPE_LONGTEXT.equalsIgnoreCase(dataType)) {
208             return DATA_TYPE_H2_VARCHAR;
209         } else if (DATA_TYPE_INTEGER.equalsIgnoreCase(dataType)) {
210             return DATA_TYPE_H2_INTEGER;
211         } else if (DATA_TYPE_BIT.equalsIgnoreCase(dataType)) {
212             return DATA_TYPE_H2_BIT;
213         }
214         return dataType;
215     }

217     private static void executeScripts(final MultiSqlExecutor template, final List<MigrationScript> migrationScripts) {
218         for (final MigrationScript migrationScript : migrationScripts) {
219             System.out.println("==============  Executing migration script " + migrationScript.getFileName() + " ============");
220             final String sqlQuery = migrationScript.getContent().replaceAll(SCHEMA_NAME_PLACEHOLDER_REGEX, SCHEMA_NAME);
221             template.execute(sqlQuery);
222             System.out.println("==============  Finished Executing migration script " + migrationScript.getFileName() + " ============");
223         }
224     }

226     private static String getConvertedConstraintName(final String constraintType, final String constraintName) {
227         return CONNECT_TO_MYSQL && "PRIMARY KEY".equals(constraintType) ? "PRIMARY" : constraintName;
