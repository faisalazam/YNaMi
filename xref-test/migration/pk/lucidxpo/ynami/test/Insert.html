11  public abstract class Insert {
12      void insert(final MultiSqlExecutor executor,
13                  final String tableName,
14                  final List<String> columnNames,
15                  final List<Object> columnValues) {
16          final StringBuilder stringBuilder = new StringBuilder();
17  
18          final List<Object> values = newArrayList();
19          for (final Object columnValue : columnValues) {
20              if (columnValue instanceof DateTime) {
21                  values.add("to_date('" + ((DateTime) columnValue).toString("dd/MM/yyyy") + "','dd/MM/yyyy')");
22              } else if ("current_date".equals(columnValue) || columnValue instanceof Integer) {
23                  values.add(columnValue);
24              } else {
25                  values.add(strVal((String) columnValue));
26              }
27          }
28  
29          final String sqlString = stringBuilder.append("INSERT INTO ")
30                  .append(tableName)
31                  .append(" (")
32                  .append(join(columnNames, ","))
33                  .append(") ")
34                  .append("VALUES (")
35                  .append(join(values, ","))
36                  .append(")")
37                  .toString();
38  
39          executor.getTemplate().execute(sqlString);
40      }
41  
42      public static String strVal(final String columnValue) {
43          if (columnValue == null) {
44              return "null";
45          } else {
46              return "'" + columnValue + "'";
47          }
48      }
49  
50      public abstract void to(MultiSqlExecutor executor);
51  }
