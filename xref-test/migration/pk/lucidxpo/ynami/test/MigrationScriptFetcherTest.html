22  class MigrationScriptFetcherTest {
23      private MigrationScriptFetcher fetcher;
24  
25      @BeforeEach
26      void setUp() {
27          fetcher = new MigrationScriptFetcher(SCRIPT_DIRECTORY_PATH);
28      }
29  
30      @Test
31      void shouldReturnAllMigrationScriptsBeforeACertainScriptNumber() throws Exception {
32          final List<MigrationScript> migrationScripts = fetcher.allMigrationScriptsBefore(3);
33  
34          assertEquals(2, migrationScripts.size());
35  
36          assertThat(migrationScripts.get(0).getFileName(), startsWith("V001"));
37          assertThat(migrationScripts.get(1).getFileName(), startsWith("V002"));
38  
39          assertEquals(contentForFile("V001__create_baseline.sql"), migrationScripts.get(0).getContent());
40          assertEquals(contentForFile("V002__create_samples_table.sql"), migrationScripts.get(1).getContent());
41      }
42  
43      @Test
44      void shouldSkipIgnoredScriptNumber() throws Exception {
45          final List<MigrationScript> migrationScripts = fetcher.allMigrationScriptsBefore(3, 2);
46  
47          assertEquals(1, migrationScripts.size());
48  
49          assertThat(migrationScripts.get(0).getFileName(), startsWith("V001"));
50          assertEquals(contentForFile("V001__create_baseline.sql"), migrationScripts.get(0).getContent());
51      }
52  
53      @Test
54      void shouldAlwaysGetMigrationScriptInOrderBasedOnFileName() throws Exception {
55          final List<MigrationScript> migrationScripts = fetcher.allMigrationScriptsBefore(3);
56  
57          for (int index = 0; index < 2; index++) {
58              final MigrationScript script = migrationScripts.get(index);
59              assertThat("Wrong order of file fetched", index + 1, equalTo(indexOf(script)));
60          }
61      }
62  
63      @Test
64      void shouldReturnContentOfScriptWithCertainScriptNumber() throws Exception {
65          final String content = fetcher.migrationScriptContentForIndex(2);
66  
67          assertEquals(contentForFile("V002__create_samples_table.sql"), content);
68      }
69  
70      @Test
71      void shouldThrowRuntimeExceptionWhenTryingToFindScriptNumberThatDoesNotExists() {
72          assertThrows(RuntimeException.class, () -> fetcher.migrationScriptContentForIndex(9999));
73      }
74  
75      private String contentForFile(final String fileName) throws IOException {
76          final String fullPath = SCRIPT_DIRECTORY_PATH + "/" + fileName;
77          final File file = new File(fullPath);
78  
79          return readFileToString(file, forName("UTF8"));
80      }
81  
82      private int indexOf(final MigrationScript migrationScript) {
83          return parseInt(migrationScript.getFileName().substring(1, 4));
84      }
85  }
