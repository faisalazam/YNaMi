22  class DBDataPatchScriptTest {
23      private static final String SAMPLE_TABLE = "Sample";
24      private static final String STATUS_COLUMN = "active";
25  
26      private DataPatchDBMigrationCheck dataPatchMigrationCheck;
27  
28      @BeforeEach
29      void setup() {
30          final MultiSqlExecutor executor = executorForLocalMySql();
31          final DBCleaner dbCleaner = new DBCleaner(executor);
32          final DataSource dataSource = localDataSource();
33          final DataSourceTransactionManager transactionManager = new DataSourceTransactionManager(dataSource);
34          final TransactionOperations transaction = new TransactionTemplate(transactionManager);
35  
36          dataPatchMigrationCheck = new DataPatchDBMigrationCheck(dbCleaner, executor, transaction);
37      }
38  
39      @Test
40      void shouldVerifyThatSampleHasBeenActivated() throws Exception {
41          final Integer id = randomInt();
42  
43          final Operation preOperation = executor -> new InsertSample()
44                  .withId(id)
45                  .withActive(0)
46                  .to(executor);
47  
48          final Operation postOperation = new Operation() {
49              @Override
50              public void execute(final MultiSqlExecutor executor) {
51                  verifySampleStatus(executor, id);
52              }
53  
54              private void verifySampleStatus(final MultiSqlExecutor executor, final Integer sampleId) {
55                  final Map<String, Object> sampleRow = executor.getRowExtractor(SAMPLE_TABLE).getSingleRowWithId(Integer.toString(sampleId));
56                  assertThat(sampleRow.get(STATUS_COLUMN), is(true));
57              }
58          };
59  
60          dataPatchMigrationCheck.testDbMigrationWithScriptName("update_status_in_sample_table.sql", preOperation, postOperation, 3, false);
61      }
