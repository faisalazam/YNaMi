60      @Test
61      void shouldCreateBaselineSchemaTables() throws Exception {
62          final Operation preOperation = executor -> {
63              assertFalse(tableExists("hibernate_sequence", executor));
64              assertFalse(tableExists("FeatureToggles", executor));
65              assertFalse(tableExists("Users", executor));
66              assertFalse(tableExists("Roles", executor));
67              assertFalse(tableExists("UserRoles", executor));
68          };
69  
70          final Operation postOperation = executor -> {
71              assertTrue(tableExists("hibernate_sequence", executor));
72  
73              assertColumnWith(executor, "hibernate_sequence", "next_val",
74                      DATA_TYPE_BIGINT, IS_NULLABLE, CONNECT_TO_MYSQL ? 19 : 64);
75  
76              final List<Map<String, Object>> hibernateSequenceRows = executor
77                      .getRowExtractor("hibernate_sequence")
78                      .getRowsWithSpecifiedField("next_val", "0");
79              assertThat(hibernateSequenceRows.size(), is(1));
80  
81              final String featureTogglesTableName = "FeatureToggles";
82              assertTrue(tableExists(featureTogglesTableName, executor));
83              assertConstraintExistsForTable(executor, featureTogglesTableName, "PRIMARY KEY", "PK_FeatureToggles");
84              assertConstraintExistsFor(executor, featureTogglesTableName, "FEATURE_NAME", "PRI");
85              assertColumnWith(executor, featureTogglesTableName, "FEATURE_NAME", DATA_TYPE_VARCHAR, NOT_NULLABLE, 100);
86              assertColumnWith(executor, featureTogglesTableName, "FEATURE_ENABLED", DATA_TYPE_INTEGER, IS_NULLABLE);
87              assertColumnWith(executor, featureTogglesTableName, "STRATEGY_ID", DATA_TYPE_VARCHAR, IS_NULLABLE, 200);
88              assertColumnWith(executor, featureTogglesTableName, "STRATEGY_PARAMS", DATA_TYPE_VARCHAR, IS_NULLABLE, 2000);
89  
90              final String usersTableName = "Users";
91              assertTrue(tableExists(usersTableName, executor));
92              assertConstraintExistsForTable(executor, usersTableName, "PRIMARY KEY", "PK_Users");
93              assertConstraintExistsForTable(executor, usersTableName, "UNIQUE", "UK_USERS_EMAIL");
94              assertConstraintExistsForTable(executor, usersTableName, "UNIQUE", "UK_USERS_USERNAME");
95              assertConstraintExistsFor(executor, usersTableName, "id", "PRI");
96              assertConstraintExistsFor(executor, usersTableName, "email", "UNI");
97              assertConstraintExistsFor(executor, usersTableName, "username", "UNI");
98              assertColumnWith(executor, usersTableName, "id", DATA_TYPE_VARCHAR, NOT_NULLABLE, 50);
99              assertColumnWith(executor, usersTableName, "name", DATA_TYPE_VARCHAR, NOT_NULLABLE, 40);
100             assertColumnWith(executor, usersTableName, "username", DATA_TYPE_VARCHAR, NOT_NULLABLE, 40);
101             assertColumnWith(executor, usersTableName, "email", DATA_TYPE_VARCHAR, NOT_NULLABLE, 40);
102             assertColumnWith(executor, usersTableName, "password", DATA_TYPE_VARCHAR, NOT_NULLABLE, 100);
103             assertAuditColumns(executor, usersTableName);
104 
105             final String rolesTableName = "Roles";
106             assertTrue(tableExists(rolesTableName, executor));
107             assertConstraintExistsForTable(executor, rolesTableName, "PRIMARY KEY", "PK_Roles");
108             assertConstraintExistsForTable(executor, rolesTableName, "UNIQUE", "UK_ROLES_NAME");
109             assertConstraintExistsFor(executor, rolesTableName, "id", "PRI");
110             assertConstraintExistsFor(executor, rolesTableName, "name", "UNI");

112             assertColumnWith(executor, rolesTableName, "name", DATA_TYPE_VARCHAR, NOT_NULLABLE, 60);
113             assertAuditColumns(executor, rolesTableName);
114 
115             assertThat(executor.getRowExtractor(rolesTableName).getRowsWithSpecifiedField("name", "ROLE_USER").size(), is(1));
116             assertThat(executor.getRowExtractor(rolesTableName).getRowsWithSpecifiedField("name", "ROLE_ADMIN").size(), is(1));
117             assertThat(executor.getRowExtractor(rolesTableName).getRowsWithSpecifiedField("name", "ROLE_SUPPORT").size(), is(1));

119             final String userRolesTableName = "UserRoles";

123             assertConstraintExistsForTable(executor, userRolesTableName, "FOREIGN KEY", "FK_USER_ROLES_USER_ID");

125             assertColumnWith(executor, userRolesTableName, "roleId", DATA_TYPE_VARCHAR, NOT_NULLABLE, 50);
126         };
