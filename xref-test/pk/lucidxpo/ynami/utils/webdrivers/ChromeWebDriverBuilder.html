9   import static io.github.bonigarcia.wdm.WebDriverManager.chromedriver;
10  import static java.lang.System.getenv;
11  import static java.time.Duration.ofSeconds;
12  
13  public class ChromeWebDriverBuilder {
14      private final static int TIMEOUT = 10;
15      private static final String CI = "CI";
16      private static final String HEADLESS = "--headless";
17      private static final String TEST_TYPE = "--test-type";
18      private static final String ENVIRONMENT = "environment";
19      private static final String NO_SANDBOX = "--no-sandbox";
20      private static final String START_MAXIMIZED = "--start-maximized";
21      private static final String DISABLE_DEV_SHM_USAGE = "--disable-dev-shm-usage";
22  
23      public static final String HEADLESS_MODE = "headless_mode";
24  
25      private final ChromeOptions options;
26  
27      public ChromeWebDriverBuilder() {
28          options = new ChromeOptions();
29          options.addArguments(
30                  TEST_TYPE,
31                  NO_SANDBOX,
32                  START_MAXIMIZED,
33                  DISABLE_DEV_SHM_USAGE
34          );
35      }
36  
37      public WebDriver build() {
38          final String environment = getenv(ENVIRONMENT);
39          if (CI.equalsIgnoreCase(environment)) {
40              // Tests are not passing when the webDriver is managed through WebDriverManager on Linux/local Docker
41              // container/GitHub Workflow, but all fine on local macOS. WebDriverManager is good in a way that we don't
42              // need to download/setup webdriver path, and it is doing all that fine on all systems, but for some reason
43              // it results in a NullPointerException during webElement interactions.
44              final WebDriver driver = new ChromeDriver(options);
45              // This 'implicitlyWait' is an important statement for the proper functioning of the cucumber tests on linux
46              // machine. Tests were failing only on Linux/GitHub Workflow and nothing worked even after trying like million
47              // of things. They work fine on my local Mac but may be the other machines or too fast or too slow that they
48              // keep failing there. 'implicitlyWait' is here for the rescue finally.
49              driver.manage().timeouts().implicitlyWait(ofSeconds(TIMEOUT));
50              return driver;
51          }
52  
53          final WebDriverManager webDriverManager = chromedriver();
54          webDriverManager
55                  .capabilities(options)
56                  .setup();
57          final WebDriver driver = webDriverManager.create();
58          driver.manage().timeouts().implicitlyWait(ofSeconds(TIMEOUT));
59          return driver;
60      }
61  
62      public ChromeWebDriverBuilder withAcceptInsecureCerts(final boolean acceptInsecureCerts) {
63          options.setAcceptInsecureCerts(acceptInsecureCerts);
64          return this;
65      }
66  
67      public ChromeWebDriverBuilder withSslHttpProxy(final String proxyHost, final int proxyPort) {
68          final Proxy proxy = new Proxy();
69          proxy.setSslProxy(proxyHost + ":" + proxyPort);
70          proxy.setHttpProxy(proxyHost + ":" + proxyPort);
71          options.setProxy(proxy);
72          return this;
73      }
74  
75      public ChromeWebDriverBuilder withHeadlessMode(final boolean headlessMode) {
76          if (headlessMode) {
77              options.addArguments(HEADLESS);
78          }
79          return this;
80      }
81  }
