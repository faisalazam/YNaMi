1   package pk.lucidxpo.ynami.utils.executionlisteners;
2   
3   import org.springframework.test.context.TestContext;
4   import org.springframework.test.context.TestExecutionListener;
5   
6   import static org.joda.time.DateTimeUtils.setCurrentMillisFixed;
7   import static org.joda.time.DateTimeUtils.setCurrentMillisSystem;
8   import static org.joda.time.Instant.now;
9   
10  public class TimeFreezeExecutionListener implements TestExecutionListener {
11      private static final long FROZEN_TIME = now().getMillis();
12  
13      @Override
14      public void beforeTestMethod(@SuppressWarnings("NullableProblems") final TestContext testContext) {
15          freezeTime();
16      }
17  
18      @Override
19      public void afterTestMethod(@SuppressWarnings("NullableProblems") final TestContext testContext) {
20          unFreezeTime();
21      }
22  
23      private void freezeTime() {
24          setCurrentMillisFixed(FROZEN_TIME);
25      }
26  
27      private void unFreezeTime() {
28          setCurrentMillisSystem();
29      }
30  }
