4   import org.springframework.jdbc.core.JdbcTemplate;
5   import org.springframework.test.context.TestContext;
6   import org.springframework.test.context.TestExecutionListener;

8   import javax.sql.DataSource;
9   
10  import static org.springframework.test.jdbc.JdbcTestUtils.deleteFromTables;
11  
12  public class DatabaseExecutionListener implements TestExecutionListener {
13      private static final String FLYWAY_SCHEMA_HISTORY = "flyway_schema_history";
14  
15      @Override
16      public void beforeTestClass(@SuppressWarnings("NullableProblems") final TestContext testContext) {
17          cleanDBData(testContext);
18      }
19  
20      @Override
21      public void afterTestMethod(@SuppressWarnings("NullableProblems") final TestContext testContext) {
22          cleanDBData(testContext);
23      }
24  
25      private void cleanDBData(final TestContext testContext) {
26          final DataSource dataSource = testContext.getApplicationContext().getBean(DataSource.class);
27          final JdbcTemplate jdbcTemplate = testContext.getApplicationContext().getBean(JdbcTemplate.class);
28          cleanDBData(dataSource, jdbcTemplate);
29      }
30  
31      private void cleanDBData(final DataSource dataSource, final JdbcTemplate jdbcTemplate) {
32          // The next few statements will find all the tables in our db schema
33          // Without being specific and not specifying catalog/schema names, will sometimes result in
34          // issues. For example, when root user is used to connect to db, it might also get other tables like
35          // sys_config and later on complain that it can't delete from there...
36          // Catalog/Schema names can be hard-coded as well...
37          final String schemaName = dataSource instanceof HikariDataSource
38                  ? ((HikariDataSource) dataSource).getPoolName()
39                  : null;
40  
41          // The query ["SHOW TABLES FROM " + schemaName] could have been used too, but it has different structure of the
42          // result set for both the H2 and MySql, which will add unnecessary extra logic to make it work for both.
43          final String query = "SELECT TABLE_NAME\n" +
44                  "FROM INFORMATION_SCHEMA.TABLES\n" +
45                  "WHERE TABLE_SCHEMA = '" + schemaName + "'";
46          final String[] tableNames = jdbcTemplate.queryForList(query, String.class)
47                  .stream()
48                  .filter(tableName -> !FLYWAY_SCHEMA_HISTORY.equalsIgnoreCase(tableName))
49                  .toArray(String[]::new);
50          cleanDBData(jdbcTemplate, tableNames);
51      }
52  
53      private void cleanDBData(final JdbcTemplate jdbcTemplate, final String[] tableNames) {
54          jdbcTemplate.execute("SET FOREIGN_KEY_CHECKS = 0;"); // Disable foreign key constraint checks
55          deleteFromTables(jdbcTemplate, tableNames);
56          jdbcTemplate.execute("SET FOREIGN_KEY_CHECKS = 1;"); // Enable foreign key constraint checks
57      }
58  }
