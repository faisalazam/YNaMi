17  class EntityAndDtoConversionTest {
18      private final ModelMapper modelMapper = new ModelMapper();
19  
20      @Test
21      void shouldConvertSampleEntityToSampleCreationDtoCorrectly() {
22          Sample sample = aSample().build();
23  
24          final SampleCreationDTO convertedSampleCreationDto = modelMapper.map(sample, SampleCreationDTO.class);
25          assertEquals(sample.isActive(), convertedSampleCreationDto.isActive());
26          assertEquals(sample.getAddress(), convertedSampleCreationDto.getAddress());
27          assertEquals(sample.getFirstName(), convertedSampleCreationDto.getFirstName());
28          assertEquals(sample.getLastName(), convertedSampleCreationDto.getLastName());
29      }
30  
31      @Test
32      void shouldVerifyConversionsBetweenEntityAndDtos() {
33          SampleCreationDTO sampleCreationDTO = SampleCreationDTO.builder()
34                  .active(chooseOneOf(true, false))
35                  .address(randomAlphabetic(50))
36                  .firstName(randomAlphabetic(10))
37                  .lastName(randomAlphabetic(10))
38                  .build();
39  
40          Sample convertedSample = modelMapper.map(sampleCreationDTO, Sample.class);
41          assertThat(convertedSample.isActive(), is(sampleCreationDTO.isActive()));
42          assertThat(convertedSample.getAddress(), is(sampleCreationDTO.getAddress()));
43          assertThat(convertedSample.getFirstName(), is(sampleCreationDTO.getFirstName()));
44          assertThat(convertedSample.getLastName(), is(sampleCreationDTO.getLastName()));
45  
46          SampleUpdationDTO sampleUpdationDto = SampleUpdationDTO.builder()
47                  .active(chooseOneOf(true, false))
48                  .firstName(randomAlphabetic(10))
49                  .lastName(randomAlphabetic(10))
50                  .build();
51  
52          modelMapper.map(sampleUpdationDto, convertedSample);
53          assertEquals(sampleUpdationDto.isActive(), convertedSample.isActive());
54          assertEquals(sampleCreationDTO.getAddress(), convertedSample.getAddress());
55          assertEquals(sampleUpdationDto.getFirstName(), convertedSample.getFirstName());
56          assertEquals(sampleUpdationDto.getLastName(), convertedSample.getLastName());
57  
58          convertedSample.setActive(false);
59  
60          SampleUpdateStatusDTO sampleUpdateStatusDTO = SampleUpdateStatusDTO.builder()
61                  .active(true)
62                  .build();
63  
64          modelMapper.map(sampleUpdateStatusDTO, convertedSample);
65          assertEquals(sampleUpdateStatusDTO.isActive(), convertedSample.isActive());
66          assertEquals(sampleCreationDTO.getAddress(), convertedSample.getAddress());
67          assertEquals(sampleUpdationDto.getFirstName(), convertedSample.getFirstName());
68          assertEquals(sampleUpdationDto.getLastName(), convertedSample.getLastName());
69      }
70  }
