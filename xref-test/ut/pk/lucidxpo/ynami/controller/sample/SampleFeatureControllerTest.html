20  
21  @ExtendWith(MockitoExtension.class)
22  class SampleFeatureControllerTest {
23      private MockMvc mockMvc;
24  
25      @Mock
26      private FeatureManagerWrappable featureManager;
27  
28      @InjectMocks
29      private SampleFeatureController sampleFeatureController;
30  
31      @BeforeEach
32      void setup() {
33          mockMvc = standaloneSetup(sampleFeatureController).build();
34      }
35      // =========================== Verify Feature Toggles are working as expected ===========================
36  
37      @Test
38      void shouldExpectEnabledMessageWhenCorrespondingFeatureToggleIsEnabled() throws Exception {
39          given(featureManager.isActive(CONDITIONAL_STATEMENTS_EXECUTION)).willReturn(true);
40  
41          mockMvc.perform(get("/feature-test"))
42                  .andExpect(status().isOk())
43                  .andExpect(forwardedUrl("welcome"))
44                  .andExpect(model().attribute("message", CONDITIONAL_STATEMENTS_EXECUTION.name() + " is enabled."))
45                  .andReturn();
46      }
47  
48      @Test
49      void shouldExpectDisabledMessageWhenCorrespondingFeatureToggleIsNotEnabled() throws Exception {
50          given(featureManager.isActive(CONDITIONAL_STATEMENTS_EXECUTION)).willReturn(false);
51  
52          mockMvc.perform(get("/feature-test"))
53                  .andExpect(status().isOk())
54                  .andExpect(forwardedUrl("welcome"))
55                  .andExpect(model().attribute("message", CONDITIONAL_STATEMENTS_EXECUTION.name() + " is disabled."))
56                  .andReturn();
57      }
58  }
