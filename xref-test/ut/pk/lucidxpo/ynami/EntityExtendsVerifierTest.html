20  class EntityExtendsVerifierTest {
21  
22      @Test
23      void shouldVerifyThatAllTheEntitiesAreExtendedFromAuditable() {
24          final Set<Class<?>> entityClasses = getTypesAnnotatedWith(Entity.class, BASE_PACKAGE + ".persistence.model");
25  
26          //The entity classes that don't need to be audited, should not be extended from "Auditable".
27          final Set<String> excludedEntityNames = newHashSet(
28                  AuditEntry.class.getName(),
29                  AuditEntryArchive.class.getName()
30          );
31  
32          assertThat("The list of entity classes must not be empty", entityClasses.isEmpty(), is(false));
33          assertThat(entityClasses.size() > excludedEntityNames.size(), is(true));
34  
35          for (Class<?> entityClazz : entityClasses) {
36              if (excludedEntityNames.contains(entityClazz.getName())) {
37                  assertThat(entityClazz.getName() + " should not be extended from " + Auditable.class.getSimpleName() + ".",
38                          getSuperTypes(entityClazz), not(hasItem(Auditable.class))
39                  );
40              } else {
41                  assertThat(entityClazz.getName() + " should be extended from " + Auditable.class.getSimpleName() + ".",
42                          getSuperTypes(entityClazz), hasItem(Auditable.class)
43                  );
44              }
45          }
46      }
47  }
