27  class DatabaseExecutionListenerUsageVerifierTest {
28  
29      @Test
30      void shouldVerifyThatIntegrationTestsChangingStateOfFeatureTogglesHaveDatabaseExecutionListenerAsTheirTestExecutionListeners() throws Exception {
31          final Method activatedMethod = FeatureManagerWrappable.class.getDeclaredMethod("activate", Feature.class);
32          final Method deactivatedMethod = FeatureManagerWrappable.class.getDeclaredMethod("deactivate", Feature.class);
33  
34          final Reflections reflections = new Reflections(
35                  new ConfigurationBuilder()
36                          .forPackages(BASE_PACKAGES)
37                          .setScanners(new MemberUsageScanner())
38                          .filterInputsBy(className -> requireNonNull(className).endsWith("IntegrationTest.class"))
39          );
40          final Collection<Member> usages = reflections.getMemberUsage(activatedMethod);
41          usages.addAll(reflections.getMemberUsage(deactivatedMethod));
42  
43          assertThat("The list of integration test classes changing the feature toggles' state must not be empty",
44                  usages.isEmpty(), is(false)
45          );
46          usages.stream().map(Member::getDeclaringClass).forEach(this::assertDatabaseExecutionListenerUsage);
47      }
48  
49      @Test
50      void shouldVerifyThatIntegrationTestsAnnotatedWithSqlHaveDatabaseExecutionListenerAsTheirTestExecutionListeners() {
51          final Reflections reflections = new Reflections(
52                  new ConfigurationBuilder()
53                          .forPackages(BASE_PACKAGES)
54                          .filterInputsBy(className -> requireNonNull(className).endsWith("IntegrationTest.class"))
55          );
56          final Set<Class<?>> testClassesAnnotatedWithSql = reflections.getTypesAnnotatedWith(Sql.class);
57  
58          assertThat("The list of integration test classes annotated with @Sql must not be empty",
59                  testClassesAnnotatedWithSql.isEmpty(), is(false)
60          );
61          testClassesAnnotatedWithSql.forEach(this::assertDatabaseExecutionListenerUsage);
62      }
63  
64      private void assertDatabaseExecutionListenerUsage(final Class<?> testClazz) {
65          final TestExecutionListeners testExecutionListeners = findAnnotation(testClazz, TestExecutionListeners.class);
66          assertNotNull(testExecutionListeners,
67                  format("%s should be annotated with %s with value %s",
68                          testClazz.getSimpleName(),
69                          TestExecutionListeners.class.getSimpleName(),
70                          DatabaseExecutionListener.class.getSimpleName()
71                  )
72          );
73          assertThat(testExecutionListeners.listeners(), hasItemInArray(DatabaseExecutionListener.class));
74      }
