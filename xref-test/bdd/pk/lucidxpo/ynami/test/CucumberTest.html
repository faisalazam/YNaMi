30      // example numbers are reported. This can make for hard to read reports. By using
31      // cucumber.junit-platform.naming-strategy=long, Cucumber will include the feature name in the scenario name. This
32      // makes the test results legible.
33      static final String NAMING_STRATEGY = "long";
