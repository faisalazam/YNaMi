1   package bdd.pk.lucidxpo.ynami.webdriver.hooks;

3   import bdd.pk.lucidxpo.ynami.annotations.LazyAutowired;
4   import bdd.pk.lucidxpo.ynami.config.AbstractSteps;
5   import io.cucumber.java.After;
6   import io.cucumber.java.Before;
7   import io.fluentlenium.adapter.IFluentAdapter;
8   import org.openqa.selenium.WebDriver;
9   import org.springframework.context.ApplicationContext;
10  
11  import java.util.Collection;
12  import java.util.Map;
13  
14  /**
15   * This class is responsible for managing the {@link io.fluentlenium} related hooks.
16   */
17  public class IFluentAdapterHooks {
18      private final ApplicationContext applicationContext;
19  
20      @LazyAutowired
21      public IFluentAdapterHooks(final ApplicationContext applicationContext) {
22          this.applicationContext = applicationContext;
23      }
24  
25      /**
26       * There is a {@link IFluentAdapter#initFluent(WebDriver)} method which has to be hooked with {@link Before} in
27       * order to init fluent with {@link WebDriver}.
28       * <p>
29       * One option is to hook it up here and the other option is to add {@link Before} hook in all the *Steps classes
30       * in {@link bdd.pk.lucidxpo.ynami.steps}.
31       */
32      @Before
33      public void before() {
34          final WebDriver webDriver = this.applicationContext.getBean(WebDriver.class);
35          getFluentAdapters().forEach(iFluentAdapter -> iFluentAdapter.initFluent(webDriver));
36      }
37  
38      /**
39       * There is a {@link IFluentAdapter#releaseFluent()} method which has to be called before/after
40       * {@link WebDriver#quit()} method in order to properly release/close/quit the {@link WebDriver}.
41       * <p>
42       * We need to invoke the {@link IFluentAdapter#releaseFluent()} method because we are using {@link io.fluentlenium}
43       * in this project.
44       * <p>
45       * One option is to hook it up here and the other option is to add {@link After} hook in all the *Steps classes
46       * in {@link bdd.pk.lucidxpo.ynami.steps}.
47       */
48      @After
49      public void afterScenario() {
50          getFluentAdapters().forEach(IFluentAdapter::releaseFluent);
51      }
52  
53      /**
54       * All of the *Steps classes (in {@link bdd.pk.lucidxpo.ynami.steps}) are extended from {@link AbstractSteps},
55       * and one of the parents of the {@link AbstractSteps} class is implementing the {@link IFluentAdapter} interface.
56       * <p>
57       * We can invoke the {@link ApplicationContext#getBeansOfType(Class)} with either {@link AbstractSteps} or
58       * {@link IFluentAdapter} which will result in same set of classes (i.e. a {@link Collection} containing
59       * all out *Steps classes.
60       * <p>
61       * This {@link Collection<IFluentAdapter>} will be used later to invoke {@link IFluentAdapter#initFluent(WebDriver)}
62       * and {@link IFluentAdapter#releaseFluent()} methods.
63       */
64      private Collection<IFluentAdapter> getFluentAdapters() {
65          final Map<String, IFluentAdapter> fluentAdapters = this.applicationContext.getBeansOfType(IFluentAdapter.class);
66          return fluentAdapters.values();
67      }
68  }
