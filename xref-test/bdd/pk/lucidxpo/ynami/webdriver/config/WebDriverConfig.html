7   import org.openqa.selenium.WebDriver;
8   import org.openqa.selenium.firefox.FirefoxOptions;
9   import org.openqa.selenium.htmlunit.HtmlUnitDriver;
10  import org.springframework.boot.autoconfigure.condition.ConditionalOnMissingBean;
11  import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
12  import org.springframework.context.annotation.Primary;
13  import org.springframework.context.annotation.Profile;
14  import pk.lucidxpo.ynami.utils.webdrivers.ChromeWebDriverBuilder;
15  
16  import static io.github.bonigarcia.wdm.WebDriverManager.edgedriver;
17  import static io.github.bonigarcia.wdm.WebDriverManager.firefoxdriver;
18  import static java.lang.Boolean.TRUE;
19  import static java.lang.Boolean.valueOf;
20  import static java.lang.System.getenv;
21  import static java.time.Duration.ofSeconds;
22  import static pk.lucidxpo.ynami.utils.webdrivers.ChromeWebDriverBuilder.HEADLESS_MODE;
23  
24  @Profile("!grid")
25  @LazyConfiguration
26  public class WebDriverConfig {
27      private static final String EDGE = "edge";
28      private static final String PROXY = "proxy";
29      private static final String CHROME = "chrome";
30      private static final String BROWSER = "browser";
31      private static final String FIREFOX = "firefox";
32      private static final String HTMLUNIT = "htmlunit";
33      private static final String NO_PROXY_VAR = "no_proxy-var";
34  
35      @Primary
36      @WebdriverBeanScope
37      @ConditionalOnProperty(name = BROWSER, havingValue = FIREFOX)
38      public WebDriver firefoxDriver() {
39          final Proxy proxy = new Proxy();
40          proxy.setAutodetect(false);
41          proxy.setNoProxy(NO_PROXY_VAR);
42  
43          final FirefoxOptions options = new FirefoxOptions();
44          options.setCapability(PROXY, proxy);
45  
46          final WebDriverManager webDriverManager = firefoxdriver();
47          webDriverManager
48                  .capabilities(options)
49                  .setup();
50          return webDriverManager.create();
51      }
52  
53      @Primary
54      @WebdriverBeanScope
55      @ConditionalOnProperty(name = BROWSER, havingValue = EDGE)
56      public WebDriver edgeDriver() {
57          final WebDriverManager webDriverManager = edgedriver();
58          webDriverManager.setup();
59          return webDriverManager.create();
60      }
61  
62      @Primary
63      @WebdriverBeanScope
64      @ConditionalOnProperty(name = BROWSER, havingValue = CHROME)
65      public WebDriver chromeDriver() {
66          final String headlessMode = getenv(HEADLESS_MODE);
67          final boolean runInHeadlessMode = headlessMode == null || TRUE.equals(valueOf(headlessMode));
68          return new ChromeWebDriverBuilder()
69                  .withAcceptInsecureCerts(true)
70                  .withHeadlessMode(runInHeadlessMode)
71                  .build();
72      }
73  
74      @Primary
75      @WebdriverBeanScope
76      @ConditionalOnMissingBean
77      @ConditionalOnProperty(name = BROWSER, havingValue = HTMLUNIT)
78      public WebDriver htmlUnitDriver() {
79          final WebDriver driver = new HtmlUnitDriver(true);
80          driver.manage().timeouts().implicitlyWait(ofSeconds(2));
81          return driver;
82      }
83  }
