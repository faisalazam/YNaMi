16  import java.net.URL;
17  
18  /**
19   * {@link Profile("grid")} annotation is for Selenium Grid and remotewebdriver. When we run the tests with
20   * “spring.profiles.active=grid” environment variable, the tests will use application-grid.properties file
21   * under the resources folder as the main configuration file.
22   */
23  @SuppressWarnings("JavadocReference")
24  @Profile("grid")
25  @LazyConfiguration
26  public class RemoteWebDriverConfig {
27      @Value("${selenium.grid.url}")
28      private URL url;
29  
30      @Primary
31      @WebdriverBeanScope
32      @ConditionalOnProperty(name = "browser", havingValue = "firefox")
33      public WebDriver remoteFirefoxDriver() {
34          final FirefoxOptions firefoxOptions = new FirefoxOptions();
35          return new RemoteWebDriver(this.url, firefoxOptions);
36      }
37  
38      @Primary
39      @WebdriverBeanScope
40      @ConditionalOnProperty(name = "browser", havingValue = "edge")
41      public WebDriver remoteEdgeDriver() {
42          final EdgeOptions edgeOptions = new EdgeOptions();
43          return new RemoteWebDriver(this.url, edgeOptions);
44      }
45  
46      @Primary
47      @WebdriverBeanScope
48      @ConditionalOnMissingBean
49      @ConditionalOnProperty(name = "browser", havingValue = "chrome")
50      public WebDriver remoteChromeDriver() {
51          final ChromeOptions chromeOptions = new ChromeOptions();
52          return new RemoteWebDriver(this.url, chromeOptions);
53      }
54  }
