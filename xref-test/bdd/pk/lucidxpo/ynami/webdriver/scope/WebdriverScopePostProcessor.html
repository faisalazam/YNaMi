9   import static bdd.pk.lucidxpo.ynami.webdriver.annotations.WebdriverBeanScope.WEB_DRIVER_SCOPE;
10  
11  /**
12   * This class is responsible for registering the {@link WebdriverBeanScope#WEB_DRIVER_SCOPE} custom bean scope, which
13   * has been added specifically for the management of {@link WebDriver}. The {@link WebdriverScope} is the one responsible
14   * for returning the beans annotated with {@link WebdriverBeanScope}, which in this case, will be {@link WebDriver}.
15   */
16  public class WebdriverScopePostProcessor implements BeanFactoryPostProcessor {
17      @Override
18      public void postProcessBeanFactory(ConfigurableListableBeanFactory beanFactory) throws BeansException {
19          // The first parameter, scopeName, is used to identify/specify a scope by its unique name.
20          // The second parameter, scope, is an actual instance of the custom Scope implementation that we wish to
21          // register and use.
22          beanFactory.registerScope(WEB_DRIVER_SCOPE, new WebdriverScope());
23      }
24  }
