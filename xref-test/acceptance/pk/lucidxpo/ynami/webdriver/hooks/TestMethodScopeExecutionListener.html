3   import acceptance.pk.lucidxpo.ynami.webdriver.annotations.TestMethodScopeBean;
4   import acceptance.pk.lucidxpo.ynami.webdriver.scope.TestMethodScope;
5   import org.openqa.selenium.WebDriver;
6   import org.springframework.context.ApplicationContext;
7   import org.springframework.test.context.TestContext;
8   import org.springframework.test.context.support.AbstractTestExecutionListener;
9   
10  import static java.util.Objects.requireNonNull;
11  
12  /**
13   * This listener has two responsibilities: resetting the test scope before each test and closing the Selenium driver,
14   * i.e. the browser, after each test.
15   * <p>
16   * That means the beans annotated with {@link TestMethodScopeBean} (i.e. {@link WebDriver} in this case) will be
17   * removed from the {@link ApplicationContext} and {@link TestMethodScope}, as well as {@link WebDriver} will be quit.
18   * So, each Selenium test will have its own {@link WebDriver} instance.
19   * <p>
20   * All that is achieved with the help of {@link TestMethodScopeBean}, {@link TestMethodScope} and
21   * {@link TestMethodScopeExecutionListener} classes.
22   * <p>
23   * {@link TestMethodScopeBean} will mark the beans with {@link TestMethodScopeBean#TEST_METHOD_SCOPE},
24   * {@link TestMethodScope} will act as a sort of repository for all the beans annotated with {@link TestMethodScopeBean},
25   * and finally the {@link TestMethodScopeExecutionListener} will be clearing those beans based on the lifecycle of the test.
26   */
27  public class TestMethodScopeExecutionListener extends AbstractTestExecutionListener {
28      private static final String WEB_DRIVER_BEAN_NAME = "webDriver";
29  
30      @Override
31      public void beforeTestClass(@SuppressWarnings("NullableProblems") final TestContext testContext) {
32          reset(testContext);
33      }
34  
35      @Override
36      public void afterTestMethod(@SuppressWarnings("NullableProblems") final TestContext testContext) {
37          final TestMethodScope testMethodScope = getTestMethodScopeBean(testContext);
38          if (testMethodScope.contains(WEB_DRIVER_BEAN_NAME)) {
39              ((WebDriver) requireNonNull(testMethodScope.remove(WEB_DRIVER_BEAN_NAME))).quit();
40          }
41      }
42  
43      @Override
44      public void afterTestClass(@SuppressWarnings("NullableProblems") final TestContext testContext) {
45          reset(testContext);
46      }
47  
48      private void reset(final TestContext testContext) {
49          getTestMethodScopeBean(testContext).reset();
50      }
51  
52      private TestMethodScope getTestMethodScopeBean(final TestContext testContext) {
53          return testContext.getApplicationContext().getBean(TestMethodScope.class);
54      }
55  }
