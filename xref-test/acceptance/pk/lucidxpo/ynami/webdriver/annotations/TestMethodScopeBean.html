19  /**
20   * This annotation will be used to annotate the beans which should last just for the life of the {@link Test} method,
21   * and in this case, particularly {@link WebDriver} beans.
22   * <p>
23   * That means the beans annotated with {@link TestMethodScopeBean} (i.e. {@link WebDriver} in this case) will be
24   * removed from the {@link ApplicationContext} and {@link TestMethodScope}, as well as {@link WebDriver} will be quit.
25   * So, each Selenium test will have its own {@link WebDriver} instance.
26   * <p>
27   * All that is achieved with the help of {@link TestMethodScopeBean}, {@link TestMethodScope} and
28   * {@link TestMethodScopeExecutionListener} classes.
29   * <p>
30   * {@link TestMethodScopeBean} will mark the beans with {@link TestMethodScopeBean#TEST_METHOD_SCOPE},
31   * {@link TestMethodScope} will act as a sort of repository for all the beans annotated with {@link TestMethodScopeBean},
32   * and finally the {@link TestMethodScopeExecutionListener} will be clearing those beans based on the lifecycle of the test.
33   */
34  @Bean
35  @Documented
36  @Target({METHOD})
37  @Retention(RUNTIME)
38  @Scope(TEST_METHOD_SCOPE)
39  public @interface TestMethodScopeBean {
40      String TEST_METHOD_SCOPE = "testMethodScope";
41  }
