14  /**
15   * Out of the box, Spring provides two standard bean scopes (“singleton” and “prototype”) that can be used in any
16   * Spring application, plus three additional bean scopes (“request”, “session”, and “globalSession”) for use only
17   * in web-aware applications.
18   * <p>
19   * But we need some scope just for the life of test method and hence the {@link TestMethodScopeBean}.
20   * <p>
21   * The test scope implementation is trivial. It holds a map where test scoped beans are cached,
22   * ensuring that only one instance is created. It also has a reset method for clearing the cache before each test run.
23   * <p>
24   * It works in conjunction with {@link TestMethodScopeBean}, {@link TestMethodScope} and
25   * {@link TestMethodScopeExecutionListener} classes.
26   * <p>
27   * {@link TestMethodScopeBean} will mark the beans with {@link TestMethodScopeBean#TEST_METHOD_SCOPE},
28   * {@link TestMethodScope} will act as a sort of repository for all the beans annotated with {@link TestMethodScopeBean},
29   * and finally the {@link TestMethodScopeExecutionListener} will be clearing those beans based on the lifecycle of the test.
30   */
31  @SuppressWarnings("NullableProblems")
32  public class TestMethodScope implements Scope {
33      /**
34       * NOTE the use of {@link Collections#synchronizedMap}, that is to ensure that the implementation is thread-safe
35       * because scopes can be used by multiple bean factories at the same time.
36       */
37      private final Map<String, Object> cache = synchronizedMap(newHashMap());
38      private final Map<String, Runnable> destructionCallbacks = synchronizedMap(newHashMap());
39  
40      public void reset() {
41          cache.clear();
42      }
43  
44      /**
45       * This method checks to see if the named object is there in our map. If it is, we return it, and if not, we use
46       * the ObjectFactory to create a new object, add it to our map, and return it.
47       */
48      @Override
49      public Object get(final String name, final ObjectFactory<?> objectFactory) {
50          if (!cache.containsKey(name)) {
51              cache.put(name, objectFactory.getObject());
52          }
53          return cache.get(name);
54      }
55  
56      public boolean contains(final String beanName) {
57          return cache.containsKey(beanName);
58      }
59  
60      @Override
61      public Object remove(final String name) {
62          return cache.remove(name);
63      }
64  
65      @Override
66      public void registerDestructionCallback(final String name, final Runnable callback) {
67          destructionCallbacks.put(name, callback);
68      }
69  
70      @Override
71      public Object resolveContextualObject(final String key) {
72          return null;
73      }
74  
75      @Override
76      public String getConversationId() {
77          return null;
78      }
79  }
