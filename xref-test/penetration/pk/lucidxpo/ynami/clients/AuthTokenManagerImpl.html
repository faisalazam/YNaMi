3   import lombok.Getter;
4   import lombok.Setter;
5   import org.openqa.selenium.Cookie;
6   import org.openqa.selenium.WebDriver;
7   
8   import java.util.Map;
9   import java.util.Objects;
10  
11  import static java.util.stream.Collectors.toMap;
12  import static org.openqa.selenium.Cookie.Builder;
13  import static penetration.pk.lucidxpo.ynami.config.Config.getInstance;
14  
15  @Getter
16  @Setter
17  public class AuthTokenManagerImpl implements AuthTokenManager {
18      private WebDriver driver;
19  
20      public AuthTokenManagerImpl(final WebDriver driver) {
21          this.driver = driver;
22      }
23  
24      public Cookie getCookieByName(final String name) {
25          return this.driver.manage().getCookieNamed(name);
26      }
27  
28      public void getUrl(final String url) {
29          driver.get(url);
30      }
31  
32      @Override
33      public Map<String, String> getAuthTokens() {
34          return getInstance().getSessionIDs().stream()
35                  .map(name -> driver.manage().getCookieNamed(name))
36                  .filter(Objects::nonNull)
37                  .collect(toMap(Cookie::getName, Cookie::getValue, (a, b) -> b));
38      }
39  
40      @Override
41      public void setAuthTokens(final Map<String, String> tokens) {
42          tokens.keySet().forEach(name -> {
43              driver.manage().deleteCookieNamed(name);
44              driver.manage().addCookie(new Builder(name, tokens.get(name)).build());
45          });
46      }
47  
48      @Override
49      public void deleteAuthTokens() {
50          driver.manage().deleteAllCookies();
51      }
52  }
