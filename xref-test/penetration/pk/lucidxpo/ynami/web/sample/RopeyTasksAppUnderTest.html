4   import penetration.pk.lucidxpo.ynami.behaviours.ILogin;
5   import penetration.pk.lucidxpo.ynami.behaviours.ILogout;
6   import penetration.pk.lucidxpo.ynami.behaviours.INavigable;
7   import penetration.pk.lucidxpo.ynami.model.Credentials;
8   import penetration.pk.lucidxpo.ynami.model.UserPassCredentials;
9   import penetration.pk.lucidxpo.ynami.web.WebApplication;
10  
11  import static org.openqa.selenium.By.id;
12  import static org.openqa.selenium.By.linkText;
13  import static org.openqa.selenium.By.name;
14  import static penetration.pk.lucidxpo.ynami.config.Config.getInstance;
15  
16  public class RopeyTasksAppUnderTest extends WebApplication implements ILogin, ILogout, INavigable {
17      @Override
18      public void navigate() {
19          openLoginPage();
20          login(getInstance().getDefaultCredentials());
21          viewProfile();
22          search("test");
23      }
24  
25      @Override
26      public void openLoginPage() {
27          driver.get(getInstance().getBaseUrl() + "user/login");
28          findAndWaitForElement(id("username"));
29      }
30  
31      @Override
32      public void login(final Credentials credentials) {
33          final UserPassCredentials userPassCredentials = new UserPassCredentials(credentials);
34          driver.findElement(id("username")).clear();
35          driver.findElement(id("username")).sendKeys(userPassCredentials.getUsername());
36          driver.findElement(id("password")).clear();
37          driver.findElement(id("password")).sendKeys(userPassCredentials.getPassword());
38          driver.findElement(name("_action_login")).click();
39      }
40  
41      @Override
42      public boolean isLoggedIn() {
43          driver.get(getInstance().getBaseUrl() + "task/list");
44          return driver.getPageSource().contains("Tasks");
45      }
46  
47      @Override
48      public void logout() {
49          driver.findElement(linkText("Logout")).click();
50      }
51  
52      public void viewAllUsers() {
53          driver.get(getInstance().getBaseUrl() + "admin/list");
54      }
55  
56      public void viewBobsProfile() {
57          viewProfile();
58      }
59  
60      public void viewAlicesProfile() {
61          viewProfile();
62      }
63  
64      private void viewProfile() {
65          driver.findElement(linkText("Profile")).click();
66      }
67  
68      @SuppressWarnings("SameParameterValue")
69      private void search(final String query) {
70          findAndWaitForElement(linkText("Tasks")).click();
71          driver.findElement(id("q")).clear();
72          driver.findElement(id("q")).sendKeys(query);
73          final WebElement searchBtn = driver.findElement(name("search-button"));
74          searchBtn.click();
75      }
76  }
