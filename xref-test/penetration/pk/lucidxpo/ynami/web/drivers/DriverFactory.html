60      // Return the desired driver and clear all its cookies
61      private static WebDriver getDriver(final String type, final boolean isProxyDriver) {
62          final WebDriver retVal = getInstance().findOrCreate(type, isProxyDriver);
63          try {
64              if (!retVal.getCurrentUrl().equals("about:blank")) {
65                  retVal.manage().deleteAllCookies();
66              }
67          } catch (final Exception e) {
68              log.error(e.getMessage());
69              e.printStackTrace();
70          }
71          return retVal;
72      }

75      public static void quitAll() {
76          log.debug("closing all webDrivers");
77          try {
78              if (driver != null) {
79                  driver.quit();
80              }
81              if (proxyDriver != null) {
82                  proxyDriver.quit();
83              }
84          } catch (final Exception e) {
85              log.error("Error quitting webDriver: " + e.getMessage());
86              e.printStackTrace();
87          }
88      }

90      /*
91       * Re-use drivers to reduce startup times
92       */
93      private WebDriver findOrCreate(final String type, final boolean isProxyDriver) {
94          if (isProxyDriver) {
95              if (proxyDriver != null) {
96                  return proxyDriver;
97              }
98              proxyDriver = createProxyDriver(type);
99              return proxyDriver;
100         } else {
101             if (driver != null) {
102                 return driver;
103             }
104             driver = createDriver(type);
105             return driver;
106         }
107     }

109     private WebDriver createDriver(final String type) {
110         if (CHROME.equalsIgnoreCase(type)) {
111             return createChromeDriver(new DesiredCapabilities());
112         } else if (FIREFOX.equalsIgnoreCase(type)) {
113             return createFirefoxDriver(null);
114         } else if (HTMLUNIT.equalsIgnoreCase(type)) {
115             return createHtmlUnitDriver(null);
116         }
117         throw new RuntimeException("Unsupported WebDriver browser: " + type);
118     }

120     private WebDriver createProxyDriver(final String type) {
121         if (CHROME.equalsIgnoreCase(type)) {
122             return createChromeDriver(createProxyCapabilities(CHROME));
123         } else if (FIREFOX.equalsIgnoreCase(type)) {
124             return createFirefoxDriver(createProxyCapabilities(FIREFOX));
125         } else if (HTMLUNIT.equalsIgnoreCase(type)) {
126             return createHtmlUnitDriver(createProxyCapabilities(HTMLUNIT));
127         }
128         throw new RuntimeException("Unsupported WebDriver browser: " + type);
129     }

179     private WebDriver createFirefoxDriver(DesiredCapabilities capabilities) {
180 
181         final ProfilesIni allProfiles = new ProfilesIni();
182         FirefoxProfile myProfile = allProfiles.getProfile("WebDriver");
183         if (myProfile == null) {
184             final File ffDir = new File(getProperty("user.dir") + separator + "ffProfile");
185             if (!ffDir.exists()) {
186                 ffDir.mkdir();
187             }
188             myProfile = new FirefoxProfile(ffDir);
189         }
190         myProfile.setAcceptUntrustedCertificates(true);
191         myProfile.setAssumeUntrustedCertificateIssuer(true);
192         myProfile.setPreference("webdriver.load.strategy", "unstable");
193         final String noProxyHosts = Config.getInstance().getNoProxyHosts();
194         if (!noProxyHosts.isEmpty()) {
195             myProfile.setPreference("network.proxy.no_proxies_on", noProxyHosts);
196         }
197         if (capabilities == null) {
198             capabilities = new DesiredCapabilities();
199         }
200         capabilities.setCapability("profile", myProfile);
201 //        capabilities.setCapability(PROFILE, myProfile);
202         setProperty("webdriver.gecko.driver", Config.getInstance().getDefaultDriverPath());
203         return new FirefoxDriver(new FirefoxOptions(capabilities));
204     }

206     private DesiredCapabilities createProxyCapabilities(final String type) {
207         DesiredCapabilities capabilities = null;
208         switch (type) {
209             case CHROME -> capabilities = chrome();
210             case FIREFOX -> capabilities = firefox();
211             case HTMLUNIT -> capabilities = htmlUnit();
212             default -> {
213             }
214         }
215         final Proxy proxy = getProxy();
216         assert capabilities != null;
217         capabilities.setCapability(PROXY, proxy);
218         return capabilities;
219     }

221     private static Proxy getProxy() {
222         final Proxy proxy = new Proxy();
223         final Config instance = Config.getInstance();
224         proxy.setHttpProxy(instance.getProxyHost() + ":" + instance.getProxyPort());
225         proxy.setSslProxy(instance.getProxyHost() + ":" + instance.getProxyPort());
226         return proxy;
