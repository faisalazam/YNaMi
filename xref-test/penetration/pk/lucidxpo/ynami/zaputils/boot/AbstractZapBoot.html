49      @Override
50      public void startZap(final ZapInfo zapInfo) {
51          if (zap != null || isZapRunning(port)) {
52              log.info("ZAP is already up and running! No attempts will be made to start ZAP.");
53              return;
54          }
55  
56          try {
57              start(zapInfo);
58              waitForZapInitialization(port, zapInfo.getInitializationTimeoutInMillis());
59          } catch (final IOException e) {
60              log.error("Error starting ZAP.", e);
61          }
62      }
63  
64      @Override
65      public void stopZap() {
66          if (zap == null) {
67              return; //ZAP not running
68          }
69          try {
70              log.info("Stopping ZAP");
71              final ClientApi client = new ClientApi(host, port, API_KEY);
72              client.core.shutdown();
73              sleep(2000);
74              zap.destroy();
75          } catch (final Exception e) {
76              log.warn("Error shutting down ZAP.");
77              log.warn(e.getMessage());
78              e.printStackTrace();
79          }
80      }

82      public static String getDefaultZapOptions() {
83          final StringBuilder defaultZapOptions = new StringBuilder();
84          defaultZapOptions.append("-daemon").append(" ");
85          defaultZapOptions.append("-config api.key=").append(API_KEY).append(" ");
86          defaultZapOptions.append("-config api.incerrordetails=true").append(" ");
87  //        defaultZapOptions.append("-config proxy.ip=127.0.0.1").append(" ");
88          defaultZapOptions.append("-config scanner.threadPerHost=20").append(" ");
89          defaultZapOptions.append("-config spider.thread=10").append(" ");
90  
91          final String upstreamProxyHost = getInstance().getUpstreamProxyHost();
92          if (!upstreamProxyHost.isEmpty()) {
93              final int upstreamProxyPort = getInstance().getUpstreamProxyPort();
94              log.info("Setting upstream proxy for ZAP to: " + upstreamProxyHost + ":" + upstreamProxyPort);
95              defaultZapOptions.append("-config connection.proxyChain.enabled=true").append(" ");
96              defaultZapOptions.append("-config connection.proxyChain.port=").append(upstreamProxyPort).append(" ");
97              defaultZapOptions.append("-config connection.proxyChain.hostName=").append(upstreamProxyHost).append(" ");
98          }
99          return defaultZapOptions.toString();
100     }

105     private void start(final ZapInfo zapInfo) throws IOException {
106         final String startCommand = buildStartCommand(zapInfo);
107         final ProcessBuilder processBuilder = new ProcessBuilder(startCommand.split(" +")).inheritIO();
108         setupProcessBuilder(zapInfo, processBuilder);
109 
110         createDirectories(get(DEFAULT_ZAP_LOG_PATH));
111         processBuilder.redirectOutput(new File(DEFAULT_ZAP_LOG_PATH, DEFAULT_ZAP_LOG_FILE_NAME));
112 
113         getInstance().setProxyApi(API_KEY);
114 
115         log.info("Starting ZAP with command: {}", startCommand);
116         zap = processBuilder.start();
117     }

124     private static boolean isZapRunning(final String host, final int port) {
125         return getResponseFromZap(host, port) == HTTP_OK;
126     }
127 
128     private static int getResponseFromZap(final String host, final int port) {
129         if (host == null) {
130             return -1;
131         }
132 
133         final String url = "http://" + host + ":" + port;
134 
135         int responseCode = -1;
136         try {
137             final HttpURLConnection conn = (HttpURLConnection) new URI(url).toURL().openConnection();
138             conn.setRequestMethod(HEAD);
139             responseCode = conn.getResponseCode();
140         } catch (final ConnectException e) {
141             log.debug("ZAP could not be reached at {}:{}.", host, port);
142         } catch (final IOException e) {
143             log.error("Error trying to get a response from ZAP.", e);
144         } catch (URISyntaxException e) {
145             log.error("Error trying to create URI.", e);
146         }
147         return responseCode;
148     }

152     }

154     @SuppressWarnings("SameParameterValue")
155     private static void waitForZapInitialization(final String host, final int port, final long timeoutInMillis) {
156         final long startUpTime = currentTimeMillis();
157         do {
158             if (currentTimeMillis() - startUpTime > timeoutInMillis) {
159                 final String message = "ZAP did not start before the timeout (" + timeoutInMillis + " ms).";
160                 log.error(message);
161                 Zap.stopZap();
162                 throw new ZapInitializationTimeoutException(message);
163             }
164 
165             sleep(ZAP_INITIALIZATION_POLLING_INTERVAL_IN_MILLIS);
166             log.info("Checking if ZAP has started at {}:{}...", host, port);
167         } while (!isZapRunning(host, port));
168 
169         log.info("ZAP has started!");
170     }
171 
172     @SuppressWarnings("SameParameterValue")
173     private static void sleep(final long millis) {
174         try {
175             Thread.sleep(millis);
176         } catch (final InterruptedException e) {
177             currentThread().interrupt();
178             log.error(e.getMessage(), e);
179         }
180     }
