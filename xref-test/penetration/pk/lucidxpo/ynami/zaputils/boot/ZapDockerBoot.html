3   import lombok.extern.slf4j.Slf4j;
4   import penetration.pk.lucidxpo.ynami.zaputils.ZapInfo;
5   
6   import static java.lang.System.getProperty;
7   
8   /**
9    * Class responsible to start and stop ZAP by running ZAP's Docker image.
10   * <p>
11   * <b>Docker must be installed locally for this to work.</b>
12   * <p>
13   * This will be used when {@code zapInfo.shouldRunWithDocker} is {@code true}.
14   */
15  @Slf4j
16  class ZapDockerBoot extends AbstractZapBoot {
17      private static final String DEFAULT_DOCKER_COMMAND = "docker run --rm";
18      private static final String ZAP_IMAGE_OPTION = " -i owasp/zap2docker-stable zap.sh ";
19  
20      private static final String CAS_AUTH_SCRIPTS_DEFAULT_DOCKER_PATH = "/zap/scripts/";
21      private static final String CAS_AUTH_SCRIPTS_SRC_PATH = getProperty("user.dir") + "/src/test/resources/zap/scripts/";
22  
23      ZapDockerBoot(final int port, final String host) {
24          super(port, host);
25      }
26  
27      @Override
28      String buildStartCommand(final ZapInfo zapInfo) {
29          final StringBuilder startCommand = new StringBuilder(DEFAULT_DOCKER_COMMAND);
30          appendVolumeOption(startCommand);
31          appendPortOption(startCommand);
32          startCommand.append(ZAP_IMAGE_OPTION);
33  
34          final String options = zapInfo.getOptions();
35          startCommand.append(options != null ? options : getDefaultZapOptions());
36          startCommand.append(" -port ").append(port);
37          startCommand.append(" -host ").append("0.0.0.0");
38          startCommand.append(" -config ").append("api.addrs.addr.name=.*");
39          startCommand.append(" -config ").append("api.addrs.addr.regex=true");
40  
41          return startCommand.toString();
42      }
43  
44      private void appendVolumeOption(final StringBuilder startCommand) {
45          startCommand.append(" -v ");
46          startCommand.append(CAS_AUTH_SCRIPTS_SRC_PATH);
47          startCommand.append(":");
48          startCommand.append(CAS_AUTH_SCRIPTS_DEFAULT_DOCKER_PATH);
49          startCommand.append(":ro");
50      }
51  
52      private void appendPortOption(final StringBuilder startCommand) {
53          startCommand.append(" -p ");
54          startCommand.append(port);
55          startCommand.append(":");
56          startCommand.append(port);
57      }
58  }
