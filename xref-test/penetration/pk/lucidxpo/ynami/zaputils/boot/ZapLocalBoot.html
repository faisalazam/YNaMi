16  /**
17   * Class responsible to start and stop ZAP locally.
18   * <p>
19   * <b>ZAP must be installed locally for this to work.</b>
20   * <p>
21   * This will normally be used when ZAP's {@code zapInfo.path} was provided and {@code shouldRunWithDocker} is false.
22   */
23  @Slf4j
24  class ZapLocalBoot extends AbstractZapBoot {
25  
26      ZapLocalBoot(final int port, final String host) {
27          super(port, host);
28      }
29  
30      @Override
31      void setupProcessBuilder(final ZapInfo zapInfo, final ProcessBuilder processBuilder) {
32          processBuilder.directory(getZapWorkingDirectory(zapInfo));
33      }
34  
35      @Override
36      String buildStartCommand(final ZapInfo zapInfo) {
37          final StringBuilder startCommand = new StringBuilder();
38  
39          startCommand.append("java").append(" ");
40          startCommand.append(zapInfo.getJmvOptions()).append(" ");
41          startCommand.append("-jar").append(" ");
42  
43          try {
44              final String zapJarName = retrieveZapJarName(zapInfo.getPath());
45              startCommand.append(zapJarName).append(" ");
46          } catch (final IOException e) {
47              log.error("Error retrieving ZAP's JAR file.");
48          }
49  
50          final String options = zapInfo.getOptions();
51          startCommand.append(options != null ? options : getDefaultZapOptions());
52          startCommand.append(" -host ").append(host);
53          startCommand.append(" -port ").append(port);
54          startCommand.append(" -dir ").append(DEFAULT_ZAP_TMP_PATH);
55  
56          return startCommand.toString();
57      }
58  
59      private static String retrieveZapJarName(final String path) throws IOException {
60          final Path zapPath = get(path);
61          if (isJarFile(zapPath)) {
62              final String filename = zapPath.getFileName().toString();
63              log.debug("ZapPath points to the Jar file {}", filename);
64              return filename;
65          }
66  
67          log.debug("ZapPath points to the folder {}", zapPath.getFileName().toString());
68  
69          for (final Path p : newDirectoryStream(zapPath)) {
70              if (isJarFile(p)) {
71                  final String filename = p.getFileName().toString();
72                  log.debug("Chosen Zap Jar file {}", filename);
73                  return filename;
74              }
75          }
76  
77          throw new ZapInitializationException("ZAP's JAR file was not found.");
78      }
79  
80      private static boolean isJarFile(final Path path) {
81          if (path == null) {
82              return false;
83          }
84  
85          if (!isRegularFile(path)) {
86              return false;
87          }
88  
89          final Path fileName = path.getFileName();
90  
91          if (fileName == null) {
92              return false;
93          }
94  
95          return fileName.toString().endsWith(".jar");
96      }
97  
98      private static File getZapWorkingDirectory(final ZapInfo zapInfo) {
99          final String fullPath = zapInfo.getPath();
100         final File dir = new File(fullPath);
101         if (dir.isDirectory()) {
102             return dir;
103         }
104 
105         if (dir.isFile()) {
106             return dir.getParentFile();
107         }
108 
109         return new File(getProperty("user.dir"));
110     }
111 }
