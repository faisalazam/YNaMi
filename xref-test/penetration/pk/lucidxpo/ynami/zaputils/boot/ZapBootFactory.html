3   import penetration.pk.lucidxpo.ynami.zaputils.ZapInfo;
4   
5   import static org.apache.commons.lang3.StringUtils.isNotBlank;
6   
7   /**
8    * Factory to create the correct {@link ZapBoot} implementation
9    * based on the given {@link ZapInfo} instance.
10   */
11  final class ZapBootFactory {
12  
13      static ZapBoot makeZapBoot(final ZapInfo zapInfo) {
14          if (zapInfo.shouldRunWithDocker()) {
15              return new ZapDockerBoot(zapInfo.getPort(), zapInfo.getHost());
16          }
17          if (isNotBlank(zapInfo.getPath())) {
18              return new ZapLocalBoot(zapInfo.getPort(), zapInfo.getHost());
19          }
20          return new ZapNilBoot(zapInfo.getPort(), zapInfo.getHost());
21      }
22  
23      private ZapBootFactory() {
24      }
25  }
