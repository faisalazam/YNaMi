5   package penetration.pk.lucidxpo.ynami.model;
6   
7   import java.util.Map;
8   
9   import static com.google.common.collect.Maps.newHashMap;
10  import static java.util.stream.Collectors.joining;
11  
12  public class Credentials {
13      Map<String, String> credentials;
14  
15      Credentials(final String... strings) {
16          credentials = stringsToMap(strings);
17      }
18  
19      String get(final String name) {
20          return credentials.get(name);
21      }
22  
23      void add(final String... strings) {
24          credentials.putAll(stringsToMap(strings));
25      }
26  
27      boolean containsKey(final String key) {
28          return credentials.keySet().stream().anyMatch(credName -> credName.equalsIgnoreCase(key));
29      }
30  
31      public String toString() {
32          return credentials.keySet().stream().map(key -> " " + key + "=" + credentials.get(key) + "\n").collect(joining());
33      }
34  
35      private Map<String, String> stringsToMap(final String... strings) {
36          final Map<String, String> map = newHashMap();
37          if (strings.length % 2 > 0) {
38              throw new RuntimeException("Credentials must be provided in pairs, e.g. 'username','bob'");
39          }
40  
41          for (int it = 0; it <= strings.length / 2; it = it + 2) {
42              map.put(strings[it], strings[it + 1]);
43          }
44          return map;
45      }
46  }
