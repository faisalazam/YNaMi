1   package penetration.pk.lucidxpo.ynami.steps.defs;
2   
3   import io.cucumber.java.en.Given;
4   import io.cucumber.java.en.Then;
5   import io.cucumber.java.en.When;
6   import penetration.pk.lucidxpo.ynami.model.Port;
7   import penetration.pk.lucidxpo.ynami.scanners.PortScanner;
8   
9   import java.util.List;
10  import java.util.concurrent.ExecutionException;
11  
12  import static com.google.common.collect.Lists.newArrayList;
13  import static java.util.Arrays.stream;
14  import static org.hamcrest.MatcherAssert.assertThat;
15  import static org.hamcrest.Matchers.containsInAnyOrder;
16  import static penetration.pk.lucidxpo.ynami.model.State.fromString;
17  
18  public class InfrastructureSteps {
19      private String targetHost;
20      private List<Port> portScanResults;
21      private List<Integer> selectedPorts;
22  
23      @Given("the target host name (.*)")
24      public void theTargetHostName(final String hostname) {
25          targetHost = hostname;
26      }
27  
28      @When("the (.*) ports are selected")
29      public void thePortsAreSelected(final String state) {
30          selectedPorts = newArrayList();
31          portScanResults.stream()
32                  .filter(result -> result.getState().equals(fromString(state)))
33                  .forEach(result -> selectedPorts.add(result.getNumber()));
34      }
35  
36      @Then("the ports should be (.*)")
37      public void thePortsShouldBe(final String csvPorts) {
38          final Integer[] expectedPorts = stream(csvPorts.split(",")).map(Integer::parseInt).toArray(Integer[]::new);
39          assertThat("Only the expected ports are open", selectedPorts, containsInAnyOrder(expectedPorts));
40      }
41  
42      @When("TCP ports from {int} to {int} are scanned using {int} threads and a timeout of {int} milliseconds")
43      public void scanPorts(final int from, final int to, final int threads, final int timeout) throws ExecutionException, InterruptedException {
44          final PortScanner portScanner = new PortScanner(targetHost, from, to, threads, timeout);
45          portScanResults = portScanner.scan();
46      }
47  }
