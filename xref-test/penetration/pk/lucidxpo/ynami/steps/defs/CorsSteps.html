1   package penetration.pk.lucidxpo.ynami.steps.defs;
2   
3   import io.cucumber.java.en.Then;
4   import io.cucumber.java.en.When;
5   import penetration.pk.lucidxpo.ynami.behaviours.ICors;
6   import penetration.pk.lucidxpo.ynami.web.Application;
7   
8   import static org.hamcrest.MatcherAssert.assertThat;
9   import static org.hamcrest.Matchers.equalTo;
10  
11  public class CorsSteps {
12      private Application app;
13  
14      @When("the path (.*) is requested with the HTTP method GET with the 'Origin' header set to (.*)")
15      public void thePathIsRequestedWithTheHttpMethodGetWithTheOriginHeaderSetTo(final String path, final String origin) {
16          ((ICors) app).makeCorsRequest(path, origin);
17      }
18  
19      @Then("the returned 'Access-Control-Allow-Origin' header has the value (.*)")
20      public void theReturnedAccessControlAllowOriginHeaderHasTheValue(final String origin) {
21          final String returnedHeader = ((ICors) app).getAccessControlAllowOriginHeader();
22          assertThat("The returned Access-Control-Allow-Origin header equals the Origin", returnedHeader, equalTo(origin));
23      }
24  
25      @Then("the 'Access-Control-Allow-Origin' header is not returned")
26      public void theAccessControlAllowOriginHeaderIsNotReturned() {
27          final String returnedHeader = ((ICors) app).getAccessControlAllowOriginHeader();
28          assertThat("The header 'Access-Control-Allow-Origin' header was not returned", returnedHeader, equalTo(null));
29      }
30  
31      public Application getWebApplication() {
32          return app;
33      }
34  }
